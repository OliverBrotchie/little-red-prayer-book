% Icon of Saint Andrew the First-Called: no title, centred, in a single rule
% border with a flower at the middle of the top and bottom rules. The whole
% plate is one picture, so the rule and the flowers share a coordinate system
% and need no overlay pass to line up.
\thispagestyle{empty}
\vspace*{\fill}
\begin{center}
\begin{tikzpicture}
  \node[inner sep=0pt] (icon) {\includegraphics[width=68mm]{assets/icon.jpg}};

  \draw[line width=0.5pt] ([shift={(-4mm,-4mm)}]icon.south west)
    rectangle ([shift={(4mm,4mm)}]icon.north east);

  \coordinate (toporn) at ([shift={(0,4mm)}]icon.north);
  \coordinate (botorn) at ([shift={(0,-4mm)}]icon.south);

  \floralornament{toporn}{yscale=1}
  \floralornament{botorn}{yscale=-1}
\end{tikzpicture}
\end{center}
